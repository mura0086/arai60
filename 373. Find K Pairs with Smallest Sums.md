# 373. Find K Pairs with Smallest Sums

## STEP1
```text
何もみずに5分以内に解いてみる
時間内にはnums1,2のすべての組み合わせを試すO(N^2)解法しか思いつかず。
答えをみた。
```
```cpp
class Solution {
    struct NumToSum {
        int num1, num2, sum;
        bool operator >(const NumToSum& other) const {
            return sum > other.sum;
        }
    };

    struct NumToIndex {
        int num_index1, num_index2;
        bool operator <(const NumToIndex& other) const {
            return num_index1 != other.num_index1 ? num_index1 < other.num_index1 : num_index2 < other.num_index2;
        }
    };
public:
    
    vector<vector<int>> kSmallestPairs(vector<int>& nums1, vector<int>& nums2, int k) {
        set<NumToIndex> visited;
        priority_queue<NumToSum, vector<NumToSum>, greater<NumToSum>> k_smallest_queue;
        vector<vector<int>> k_smallest_pair;
        visited.insert({0, 0});
        k_smallest_queue.push({0, 0, 0});
        while (k_smallest_pair.size() < k) {
            auto [i, j, _] = k_smallest_queue.top(); 
            k_smallest_queue.pop();
            k_smallest_pair.push_back({nums1[i], nums2[j]});
            if (i + 1 < nums1.size() && !visited.contains({i + 1, j})) {
                k_smallest_queue.push({i + 1, j, nums1[i + 1] + nums2[j]});
                visited.insert({i + 1, j});
            }
            if (j + 1 < nums2.size() && !visited.contains({i, j + 1})) {
                k_smallest_queue.push({i, j + 1, nums1[i] + nums2[j + 1]});
                visited.insert({i, j + 1});
            }
        }
        return k_smallest_pair;
    }
};
```
#### 感想
```text
以前setやpriority_queueでpairをそのまま使ってたところ
指摘があったので構造体を使用してみた。情報量が増えてよりわかりやすい
```

## STEP2
### プルリクやドキュメントを参照
- indexの変数にi,jを使ってみたがindexのほうがよい
- visitedを更新するタイミングを答えを更新するときと同じにしたほうが簡潔になる
- 処理済み、というニュアンスを含めるためにvisited->processedに変更
- queueに追加するところを関数化
- 変数名を変更

```cpp
class Solution {
    struct NumIndex1Index2 {
        int index1, index2;
        bool operator <(const NumIndex1Index2& other) const {
            return index1 != other.index1 ? index1 < other.index1 : index2 < other.index2;
        }
    };

    struct NumAndSum {
        int index1, index2, sum;
        bool operator >(const NumAndSum& other) const {
            return sum > other.sum;
        }
    };

private:
    void process_and_push(int index1, int index2, vector<int>& nums1, vector<int>& nums2, set<NumIndex1Index2>& processed,
                priority_queue<NumAndSum, vector<NumAndSum>, greater<NumAndSum>>& nums_and_sums) {
        if (index1 >= nums1.size() || index2 >= nums2.size() || processed.contains({index1, index2})) {
            return;
        }
        processed.insert({index1, index2});
        nums_and_sums.push({index1, index2, nums1[index1] + nums2[index2]});
    }

public:
    vector<vector<int>> kSmallestPairs(vector<int>& nums1, vector<int>& nums2, int k) {
        set<NumIndex1Index2> processed;
        priority_queue<NumAndSum, vector<NumAndSum>, greater<NumAndSum>> nums_and_sums;
        vector<vector<int>> k_smallest_nums;
        processed.insert({0, 0});
        nums_and_sums.push({0, 0, 0});
        while (k_smallest_nums.size() < k) {
            auto [index1, index2, _] = nums_and_sums.top();
            nums_and_sums.pop();
            k_smallest_nums.push_back({nums1[index1], nums2[index2]});
            process_and_push(index1 + 1, index2, nums1, nums2, processed, nums_and_sums);
            process_and_push(index1, index2 + 1, nums1, nums2, processed, nums_and_sums);
        } 
        return k_smallest_nums;
    }
};
```
#### 感想
```text
関数化することでメインの処理が簡潔に書けるようになった
setの変数名、意味合い的にはprocessedでいいが重複チェックという観点だとseen等のほうがわかりやすい気もしている
```
