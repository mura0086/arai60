# 695. Max Area of Island

## STEP1
```text
何もみずに解いてみる
```
### 考えたこと
- 各エリアが島かどうかチェック
- 島であればそこから島が続く限りサイズを記録しながら探索する
- 探索がおわったらその時点までの最大サイズと比較し
- 上回っていたら最大サイズを更新

```cpp
class Solution {
 public:
  int maxAreaOfIsland(vector<vector<int>>& grid) {
    const int height = grid.size();
    const int width = grid[0].size();
    vector<vector<uint8_t>> visited(height, vector<uint8_t> (width));
    int max_area_of_island = 0;
    for (int x = 0; x < height; x++) {
      for (int y = 0; y < width; y++) {
        if(grid[x][y] == kLand && !visited[x][y]) {
          int area_of_island = 0;
          traverseIsland(x, y, grid, visited, area_of_island, height, width);
          max_area_of_island = max(max_area_of_island, area_of_island);
        }
      }
    }
    return max_area_of_island;
  }
 private:
  static const int kLand = 1;
  void traverseIsland(int x, int y, const vector<vector<int>> grid, vector<vector<uint8_t>>& visited, int& area_of_island, const int height, const int width) {
    if (!(0 <= x && x < height && 0 <= y && y < width)) {
      return;
    }
    if (visited[x][y] || !grid[x][y] == kLand) {
      return;
    }
    visited[x][y] = 1;
    area_of_island += 1;
    traverseIsland(x + 1, y, grid, visited, area_of_island, height, width);
    traverseIsland(x, y + 1, grid, visited, area_of_island, height, width);
    traverseIsland(x - 1, y, grid, visited, area_of_island, height, width);
    traverseIsland(x, y - 1, grid, visited, area_of_island, height, width);
  }
};
```
### 感想
```text
スレッドセーフを考慮し、状態はメンバ変数にせず引数で渡す設計にしているが
heightやwidthのような固定された値を毎回渡すのも効率が悪い気がする。
gridのサイズをgrid.sizeのように直接設定するのも考えたが可読性が下がると感じどうすべきか悩んだ
```
##### コメントが付きそうなポイント
```text
・max_area_of_islandの更新タイミング
・area_of_islandを参照渡しにしているところ
・traverseIslandの呼び方、命名
　べた書きでなく移動先を配列に持っておいて回すなど
・kLandがあるならkWaterもあっていい
・1行の長さ
・現在の位置を表す変数x,yの置き方
```

### プルリクやドキュメントを参照
```text

・また現在の島のサイズを参照渡しすると参照透明性に関連した副作用があるとのコメントを見かけた
　STEP1のコードは参照透過性はない、visited,area_of_islandを参照渡しして更新しているが
　あくまで意図的なものなので想定された範囲内での安全な副作用である認識
　ほかにどのような影響が考えられるのか調査したが判然としなかった

・x,yを移動用変数に使用している方が見受けられなかった、前回の問題ではdirectionとしてdx,dyを用意していたので
　それに合わせていたが今回はその限りではない、かつやはりrow,columnのほうが直感的なのでこちらを使用する
```

```cpp
class Solution {
 public:
  int maxAreaOfIsland(vector<vector<int>>& grid) {
    const int num_rows = grid.size();
    const int num_columns = grid[0].size();
    int max_area_of_island = 0;
    vector<vector<uint8_t>> visited(num_rows, vector<uint8_t> (num_columns));
    for (int row = 0; row < num_rows; row++) {
      for (int column = 0; column < num_columns; column++) {
        if (!visited[row][column] && grid[row][column] == kLand) {
          max_area_of_island = max(max_area_of_island, 
                                   getAreaOfIsland(row, column, grid, visited, num_rows, num_columns));
        }
      }
    }
    return max_area_of_island;
  }
 private:
  static const int kLand = 1;
  bool isNewIsland(int row, int column, const vector<vector<int>>& grid,
                   vector<vector<uint8_t>>& visited, const int& num_rows, const int& num_columns) {
    if (!(0 <= row && row < num_rows && 0 <= column && column < num_columns)) {
      return false;
    }
    if (grid[row][column] != kLand) {
      return false;
    }
    if (visited[row][column]) {
      return false;
    }
    return true;
  }
  int getAreaOfIsland(int row, int column, const vector<vector<int>>& grid,
                      vector<vector<uint8_t>>& visited, const int& num_rows, const int& num_columns) {
    if (!isNewIsland(row, column, grid, visited, num_rows, num_columns)) {
      return 0;
    }
    visited[row][column] = 1;
    int area_of_island = 1;
    area_of_island += getAreaOfIsland(row + 1, column, grid, visited, num_rows, num_columns);
    area_of_island += getAreaOfIsland(row, column + 1, grid, visited, num_rows, num_columns);
    area_of_island += getAreaOfIsland(row - 1, column, grid, visited, num_rows, num_columns);
    area_of_island += getAreaOfIsland(row, column -1, grid, visited, num_rows, num_columns);
    return area_of_island;
  }
};
```

## STEP4
- stackを用いた非再帰DFS
- stackに追加するまえに範囲チェックを入れた
- 計算オーダーは変わらないが、定数倍の処理差が生まれる可能性がある
- 非再帰的な方法は スタックオーバーフローのリスクを減らすが、ヒープ領域に割り当てられるメモリ量には依然注意が必要
- 実際、再帰、非再帰はどのように選択するべきか今は判然としていないので都度意識していきたい。

```cpp
class Solution {
 public:
  int maxAreaOfIsland(vector<vector<int>>& grid) {
    const int num_columns = grid[0].size();
    const int num_rows = grid.size();
    vector<vector<uint8_t>> visited(num_rows, vector<uint8_t> (num_columns) );
    int max_area_of_island = 0;
    for (int row = 0; row < num_rows; row++) {
      for (int column = 0; column < num_columns; column++) {
        if (grid[row][column] == kSea || visited[row][column]) {
           continue;
        }
        max_area_of_island = max(max_area_of_island, getIslandArea(row, column, grid, visited, num_rows, num_columns));
      }      
    }
    return max_area_of_island;
  }

 private:
  static const int kSea = 0;
  int getIslandArea(int row, int column, const vector<vector<int>>& grid, vector<vector<uint8_t>>& visited, int num_rows, int num_columns) {
    stack<pair<int,int>> next_coordinates;
    next_coordinates.push({row, column});
    int area = 1;
    visited[row][column] = 1;
    while (!next_coordinates.empty()) {
      auto[row, column] = next_coordinates.top();
      next_coordinates.pop();
      auto checkAndExpand = [&](int row, int column) {
        if (!(0 <= row && row < num_rows && 0 <= column && column < num_columns)) {
          return;
        } 
        if (visited[row][column] || grid[row][column] == kSea) {
          return;
        }
        next_coordinates.push({row, column});
        area++;
        visited[row][column] = 1;
      };
      checkAndExpand(row - 1, column);
      checkAndExpand(row + 1, column);
      checkAndExpand(row, column - 1);
      checkAndExpand(row, column + 1);
    }
    return area;
  }
};
```
