# 49. Group Anagrams

## STEP1
```text
何もみずに5分以内に解いてみる
```
```cpp
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        map<string, vector<string>> string_to_anagrams;
        for (string str : strs) {
            string sorted_string = str;
            sort(sorted_string.begin(), sorted_string.end());
            string_to_anagrams[sorted_string].push_back(str);
        }
        vector<vector<string>> groupAnagrams;
        for (auto[_, anagrams] : string_to_anagrams) {
            groupAnagrams.push_back(anagrams);
        }
        return groupAnagrams; 
    }
};
```

## STEP2
### プルリクやドキュメントを参照
- string は C/C++ の予約語と被っている
- strはビルトインタイプの名前なので、変数名としては使わないことが多い。
- キーとして、アルファベットの出現頻度を用いる。

```cpp
class Solution {

    vector<vector<string>> groupAnagrams(vector<string>& strs) {

```
#### アルファベットの出現頻度をキーとする
```cpp
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        map<vector<int>, vector<string>> letter_to_frequency;
        for (string word : strs) {
            vector<int> frequency(26);
            for (char letter : word) {
                frequency[letter - 'a']++;
            }
            letter_to_frequency[frequency].push_back(word);
        }
        vector<vector<string>> groupAnagrams;
        for (auto[_, anagrams] : letter_to_frequency) {
            groupAnagrams.push_back(anagrams);
        }
        return groupAnagrams; 
    }
};
```
#### 感想
```text
mapのキーにアルファベットの出現頻度を設定する発想がいままでなかった
```
