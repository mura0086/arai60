# 200. Number of Islands

## STEP1
```text
何もみずに5分以内に解いてみる
```
#### 考えたこと
```
すべての地点を確認
未訪問かつ島である場合island_cntを+1。
その地点からdfsを実施。未訪問の島のエリアをすべて探索する。
時間計算量　O(mn)
```
```cpp
class Solution {
 public:
  vector<vector<char>> grid;
  vector<vector<bool>> visited;
  int cnt = 0;
  int column;
  int row;
  vector<int> dx = {-1, 0, 1, 0};
  vector<int> dy = {0, -1, 0, 1};
  int numIslands(vector<vector<char>> &grid) {
    this->grid = grid;
    column = grid[0].size();
    row = grid.size();
    visited.resize(row, vector<bool>(column));
    for (int i = 0; i < row; i++) {
      for (int j = 0; j < column; j++) {
        if (!visited[i][j] && grid[i][j] == '1') {
          cnt++;
          numIslandsHelper(i, j);
        }
      }
    }
    return cnt;
  }

 private:
  void numIslandsHelper(const int &x, const int &y) {
    if (!(x < row && x >= 0 && y < column && y >= 0)) {
      return;
    }
    if (visited[x][y] || grid[x][y] != '1') {
      return;
    }
    visited[x][y] = true;
    if (grid[x][y] == '1') {
      for (int i = 0; i < 4; i++) {
        numIslandsHelper(x + dx[i], y + dy[i]);
      }
    }
  }
};

```

#### bfs
```cpp
class Solution {
 public:
  struct RowToColumn {
    int row;
    int column;
  };
  int numIslands(vector<vector<char>>& grid) {
    const int width = grid[0].size();
    const int height = grid.size();
    char kSea = '0';
    char kLand = '1';
    int number_of_island = 0;
    vector<vector<uint8_t>> visited(height, vector<uint8_t>(width));
    queue<RowToColumn> next_coordinates;
    vector<int> next_row = {-1, 0, 1, 0};
    vector<int> next_column = {0, -1, 0, 1};
    auto IsIsland = [&]() {
      while(!next_coordinates.empty()) {
        auto [row, column] = next_coordinates.front();
        next_coordinates.pop();
        if (grid[row][column] == kSea) {
            continue;
        }
        if (visited[row][column]) {
            continue;
        }
        visited[row][column] = 1;
        for (int i = 0; i < 4; i++) {
          if (!(row + next_row[i] < height && row + next_row[i] >=0)) {
            continue;
          }
          if (!(column + next_column[i] < width && column + next_column[i] >= 0)) {
            continue;
          }
          next_coordinates.push({row + next_row[i], column + next_column[i]});
        }
      }
    };
    for(int i = 0; i < height; i++) {
      for (int j = 0; j < width; j++) {
        if (grid[i][j] == kLand && !visited[i][j]) {
            next_coordinates.push({i, j});
            number_of_island++;
            IsIsland();
        } 
      }
    }
    return number_of_island;
  }
};
```

```text

変更点
```

- 変数名、関数名の変更
```text
現在の座標をrow,columで表現していたので
四方への移動用配列dx,dyと統一がとれていなかった
dx,dyをdr,dcへ変更する選択肢もあったが
簡潔に感じたので現在の座標をx,yに変更した
Isislandはboolを返すニュアンスが強く適さない
島を横断するニュアンスのtraverseIslandとした
```

- メンバ変数をローカル変数に変更
```text
numIslandsとdfsの関数で受け渡しをする変数の数を多くしたくなかったことから
メンバ変数として各関数から呼び出していたが複数スレッドから同時に使ったときにお互いの状態に干渉する
可能性があったので、島か海か、という判定以外はローカル変数として定義した
島か海か、という条件は全インスタンスで共有されるクラス唯一の値であるので
privateなstatic constメンバ変数として定義とした
https://google.github.io/styleguide/cppguide.html#Constant_Names
```

```cpp
class Solution {
 public:
  int numIslands(vector<vector<char>>& grid) {
    const int height = grid.size();
    const int width = grid[0].size();
    int lands = 0;
    vector<vector<uint8_t>> visited(height, vector<uint8_t> (width));
    for (int x = 0; x < height; x++) {
      for (int y = 0; y < width; y++) {
        if (grid[x][y] == kLand && !visited[x][y]) {
            lands++;
            traverseIsland(x, y, grid, visited, height, width);
        }
      }  
    }
    return lands;
  }

 private:
  static const char kLand = '1';
  static const char kSea = '0';
  void traverseIsland(int x, int y, const vector<vector<char>>& grid, vector<vector<uint8_t>>& visited, const int height, const int width) {
    if (!(0 <= x && x < height && 0 <= y && y < width)) {
      return;
    }
    if (visited[x][y]) {
      return;
    }
    if (grid[x][y] == kSea) {
      return;
    }
    visited[x][y] = 1;
    const vector<int> dx = {-1, 0, 1, 0};
    const vector<int> dy = {0, -1, 0, 1};
    for (int i = 0; i < 4; i++) {
      traverseIsland(x + dx[i], y + dy[i], grid, visited, height, width);
    }
  }
};
```
