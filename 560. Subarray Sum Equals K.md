class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        int number_of_sum_equals_to_k = 0;
        for (int i = 0; i < nums.size(); i++) {
            int j = i;
            int subarray_sum = 0;
            while (j < nums.size()) {
                subarray_sum += nums[j];
                if (subarray_sum == k) {
                    number_of_sum_equals_to_k++;
                }
                j++;
            }
        }
        return number_of_sum_equals_to_k;
    }
};

- 累積和＋総当たり
```cpp
class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        int number_of_sum_equals_to_k = 0;
        vector<int> subarray(nums.size() + 1);
        for (int i = 1; i < nums.size() + 1; i++) {
            subarray[i] = nums[i - 1] + subarray[i - 1];
        }
        for (int i = 0; i < nums.size() + 1; i++) {
            for (int j = 0; j < i; j++) {
                if (i == j) {
                    continue;
                }
                if (subarray[i] - subarray[j] == k) {
                    number_of_sum_equals_to_k++;
                }
            }
        }
        return number_of_sum_equals_to_k;
    }
};
```
1-➂
- ハッシュマップ＋累積和
- 現在の累積和sumを更新
- sum-kが以前に登場していたら、その時のインデックスiから現在jまでの部分配列の合計はk
- sumをmapに記録（今までにこの累積和が何回出たか）
```cpp
class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int, int> prefix_sum_count;
        prefix_sum_count[0] = 1;
        int current_sum = 0;
        int number_of_sum_equals_to_k = 0;
        for (int num : nums) {
            current_sum += num;
            if (prefix_sum_count.count(current_sum - k)) {
                number_of_sum_equals_to_k += prefix_sum_count[current_sum - k];
            }
            prefix_sum_count[current_sum]++;
        }
        return number_of_sum_equals_to_k;
    }
};
```
#### 感想
```text
ハッシュマップ＋累積和の考え方は理解に少し時間がかかった
現時点での総和-kとなる地点からの配列の合計がkとなるような地点の数を回収していくイメージで理解した
```
